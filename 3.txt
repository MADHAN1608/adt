//using WHERE Clause to filter employeers whose salary is greater tham 40000

SELECT * FROM Employee WHERE Salary > 40000;

//using ORDER BY Clause (sorting the employees by salary in descing order)

SELECT * FROM Employee ORDER BY Salary;


//using the aggregate functions

//counting the total numbers of employeers

SELECT COUNT(*) AS Total_Employees FROM Employee;

//finding total salary of all employees

SELECT SUM(salary) AS Total_Salary FROM Employee;

//find average salary

SELECT AVG(Salary) as Average_Salary FROM Employee;

//finding the highest salary fo an employee

SELECT Max(Salary) AS Highest_Salary FROM Employee;

//finding the lowest salary of an employee

SELECT MIN(Salary) AS Lowest_Salary FROM Employee;
